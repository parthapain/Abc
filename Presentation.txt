<script>
  const durations = [4000, 8000, 8500, 8000, 7000, 10000, 11000, 12500, 6000]; // = 75000ms
  const scenes = Array.from(document.querySelectorAll('.scene'));
  const progressWrap = document.getElementById('progress-wrap');
  const dotnav = document.getElementById('dotnav');
  const pauseBtn = document.getElementById('pauseBtn');
  const replayBtn = document.getElementById('replayBtn');
  const sceneNumEl = document.getElementById('sceneNum');
  const watermark = document.getElementById('watermark');
  document.getElementById('sceneTotal').textContent = scenes.length;

  scenes.forEach((_, i) => {
    const seg = document.createElement('div');
    seg.className = 'seg';
    seg.addEventListener('click', () => goToScene(i));
    const fill = document.createElement('div');
    fill.className = 'seg-fill';
    seg.appendChild(fill);
    progressWrap.appendChild(seg);

    const dot = document.createElement('button');
    dot.setAttribute('aria-label', 'Go to scene ' + (i + 1));
    dot.addEventListener('click', () => goToScene(i));
    dotnav.appendChild(dot);
  });
  const fills = Array.from(document.querySelectorAll('.seg-fill'));
  const dots = Array.from(dotnav.querySelectorAll('button'));

  let current = -1;
  let timer = null;
  let paused = false;
  let segStartTime = 0;
  let segElapsed = 0;

  function clearFillAnim(el){ el.style.transition = 'none'; el.style.width = '0%'; void el.offsetWidth; }

  function goToScene(i){
    if(i >= scenes.length){ i = scenes.length - 1; }
    if(i < 0){ i = 0; }
    scenes.forEach(s => s.classList.remove('active'));
    scenes[i].classList.add('active');
    dots.forEach((d, idx) => d.classList.toggle('active', idx === i));
    sceneNumEl.textContent = i + 1;
    watermark.style.opacity = (i === 0) ? '0' : '1';
    current = i;
    segElapsed = 0;
    fills.forEach((f, idx) => {
      if(idx < i){ f.style.transition='none'; f.style.width='100%'; }
      else if(idx > i){ f.style.transition='none'; f.style.width='0%'; }
    });
    if(!paused){ runSegment(durations[i]); }
    else { clearFillAnim(fills[i]); clearTimeout(timer); }
  }

  function runSegment(duration){
    const fill = fills[current];
    clearFillAnim(fill);
    requestAnimationFrame(() => {
      fill.style.transition = `width ${duration}ms linear`;
      fill.style.width = '100%';
    });
    segStartTime = Date.now();
    clearTimeout(timer);
    timer = setTimeout(() => { if(current < scenes.length - 1){ goToScene(current + 1); } }, duration);
  }

  function togglePause(){
    if(!paused){
      paused = true;
      document.body.classList.add('is-paused');
      clearTimeout(timer);
      segElapsed += Date.now() - segStartTime;
      const fill = fills[current];
      const computedWidth = getComputedStyle(fill).width;
      fill.style.transition = 'none';
      fill.style.width = computedWidth;
      pauseBtn.textContent = '▶ Play';
    } else {
      paused = false;
      document.body.classList.remove('is-paused');
      const remaining = Math.max(durations[current] - segElapsed, 50);
      const fill = fills[current];
      segStartTime = Date.now();
      requestAnimationFrame(() => {
        fill.style.transition = `width ${remaining}ms linear`;
        fill.style.width = '100%';
      });
      timer = setTimeout(() => { if(current < scenes.length - 1){ goToScene(current + 1); } }, remaining);
      pauseBtn.textContent = '⏸ Pause';
    }
  }

  pauseBtn.addEventListener('click', togglePause);
  replayBtn.addEventListener('click', () => window.location.reload());

  goToScene(0);
</script>
